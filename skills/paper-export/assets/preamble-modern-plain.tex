\makeatletter
\fancypagestyle{peplainfirst}{%
  \fancyhf{}%
  \fancyfoot[C]{\pe@hf\pe@pagenum}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}
\renewcommand{\maketitle}{%
  \begingroup
    \thispagestyle{peplainfirst}%
    \begin{center}%
      {\heiti\zihao{2}\@title\par}%
    \end{center}%
    \vspace{12pt}%
  \endgroup
  \pagestyle{fancy}%
}
\makeatother
